// MAG CORE - Interpolation Tactique • MIT • v07
export const lerp = (a,b,t) => a + (b-a)*t
export const damp = (c,t,l=0.08,dt=1) => lerp(c,t,1-Math.exp(-l*dt))
export const edgeInterpolate = (f,to,t) => f+(to-f)*(1-Math.pow(2,-10*t))
export const smoothStep = (t) => t*t*(3-2*t)
export const map = (v,inMin,inMax,outMin,outMax) => outMin+((v-inMin)*(outMax-outMin))/(inMax-inMin)
export const spring = (c,t,vel,s=0.15,d=0.8) => { const f=(t-c)*s; const nv=(vel+f)*d; return {value:c+nv,velocity:nv} }
